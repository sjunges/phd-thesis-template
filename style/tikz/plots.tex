% Style for pgfplots

\pgfplotsset{
    %compat=1.15,
    % Set style
    general plot style/.style={
        axis x line=bottom, axis y line=left,
        x label style={at={(axis description cs:0.5,-0.05)},anchor=north},
        y label style={at={(axis description cs:0.05,0.5)},anchor=south},
        yticklabel style={font=\footnotesize}, xticklabel style={rotate=290, anchor=west, font=\footnotesize},
        legend style={font=\footnotesize},
    },
    quantile plot style/.style={
        general plot style,
        width=0.95\textwidth, height=8cm,
        ytick={0.1, 1, 10, 60, 600, 3600}, yticklabels={0.1s, 1s, 10s, 1m, 10m, 1h},
    },
    scatter plot style/.style={
        general plot style,
        width=7cm, height=7cm,
        x label style={at={(axis description cs:0.5,-0.04)},anchor=north},
        y label style={at={(axis description cs:0.10,0.5)},anchor=south},
        extra x tick labels = {\TO, \MO, \ERR},
        extra x tick style = {grid = major},
        extra y tick labels = {\TO, \MO, \ERR},
        extra y tick style = {grid = major},
    },
    scatter plot times style/.style={
        scatter plot style,
        xmin=0.05, ymin=0.05, xmax=14400, ymax=14400,
        x label style={at={(axis description cs:0.5,-0.02)},anchor=north},
        xtick={0.1, 1, 60, 600, 1800}, xticklabels={0.1, 1, 60, 600, 1800},
        ytick={0.1, 1, 60, 600, 1800}, yticklabels={0.1, 1, 60, 600, 1800},
        extra x ticks = {3600, 7200, 14400},
        extra y ticks = {3600, 7200, 14400},
    },
    scatter plot states style/.style={
        scatter plot style,
        xmin=1, ymin=1, xmax=160000000, ymax=160000000,
        xtick={10, 100, 1000, 10000, 100000, 1000000, 10000000}, xticklabels={10,100, 1k, 10k, 100k, 1m, 10m},
        ytick={10, 100, 1000, 10000, 100000, 1000000, 10000000}, yticklabels={10,100, 1k, 10k, 100k, 1m, 10m},
        extra x ticks = {40000000, 80000000, 160000000},
        extra y ticks = {40000000, 80000000, 160000000},
    },
}
